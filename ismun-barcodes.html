3Adding more delegates without starting over

Once I had generated the original set of barcodes, another problem came up: there were extra JCC internal delegates to add. I didn't want to regenerate hundreds of cards just because a few more names had arrived. That could also create confusion if someone's barcode number changed.

So I wrote a small Python appender that works with the system I'd already built. It first opens my existing id_registry.json file, checks the last barcode number used, and looks for an Excel file containing the extra delegates. It then goes through the Full name column, ignores blank entries, and skips people whose names are already in the registry.

For every new name, the script assigns the next available number and creates a Code 128 barcode card as a high-quality JPEG. Each card shows the person's name, number, and the label Internal Delegate | JCC. The program also updates the registry so I can keep track of which numbers belong to which people.

The part I particularly liked was that I could add delegates without touching the original cards. I designed it to be rerunnable for names already in the registry, which made last-minute additions much easier. This particular script handles adding barcode cards; the scanning and meal logging side is a separate part of my overall ISMUN system.

This does NOT regenerate the whole set. It reads the existing
ISMUN_BARCODES/id_registry.json, finds the highest number already used,
and appends one new barcode per NAMED person in additional_barcodes.xlsx
(rows with a blank Full name are skipped). New cards are labelled as
Internal Delegate / JCC, exactly like the JCC internal delegates already
in the set.

def _find_xlsx(*keywords):
    hits = [p for p in Path(DOWNLOADS).glob("*.xlsx")
            if not p.name.startswith("~$")
            and all(k in p.name.lower() for k in keywords)]
    if not hits:
        raise SystemExit(f"No .xlsx in {DOWNLOADS} matching {keywords}")
    hits.sort(key=lambda p: p.stat().st_mtime, reverse=True)
    return str(hits[0])


def _font(size, bold=False):
    sd = "-Bold" if bold else ""
    sl = "Bold" if bold else "Regular"
    for path in [
        "/Library/Fonts/Arial Bold.ttf" if bold else "/Library/Fonts/Arial.ttf",
        "/System/Library/Fonts/Supplemental/Arial Bold.ttf" if bold

        "C:/Windows/Fonts/arialbd.ttf" if bold else "C:/Windows/Fonts/arial.ttf",

        try:
            return ImageFont.truetype(path, size)
        except (OSError, IOError):
            continue
    try:
        return ImageFont.load_default(size=size)
    except TypeError:
        return ImageFont.load_default()


_FONTS = None


def _fonts():
    global _FONTS
    if _FONTS is None:
        _FONTS = (_font(60, True), _font(68, True), _font(44))
    return _FONTS


def make_barcode_jpg(number, name, line2, path):
    bars = barcode.get("code128", str(number), writer=ImageWriter()).render(
        {"module_width": 0.7, "module_height": 22.0, "quiet_zone": 4.0,
         "write_text": False, "dpi": 600}).convert("RGB")
    f_id, f_name, f_line2 = _fonts()
    pad = 60
    m = ImageDraw.Draw(bars)
    id_txt = f"#{number}"
    id_w = m.textlength(id_txt, font=f_id)
    name_w = m.textlength(name, font=f_name)
    line2_w = m.textlength(line2, font=f_line2)
    width = max(bars.width, int(id_w), int(name_w), int(line2_w)) + pad * 2
    height = bars.height + 360
    canvas = Image.new("RGB", (width, height), "white")
    canvas.paste(bars, ((width - bars.width) // 2, pad))
    d = ImageDraw.Draw(canvas)
    y = pad + bars.height + 24
    d.text(((width - id_w) // 2, y), id_txt, font=f_id, fill=C_ID)
    d.text(((width - name_w) // 2, y + 84), name, font=f_name, fill=C_NAME)
    d.text(((width - line2_w) // 2, y + 176), line2, font=f_line2, fill=C_LINE2)
    canvas.save(path, "JPEG", quality=JPEG_QUALITY, subsampling=0,
                dpi=(600, 600))


def _clean(s):
    return " ".join(str(s).split()).strip()


def main():
    if not os.path.exists(REGISTRY):
        raise SystemExit(f"{REGISTRY} not found — run the main generator "
                         "first.")
    with open(REGISTRY, encoding="utf-8") as f:
        registry = json.load(f)

    highest = max(int(k) for k in registry) if registry else 0
    existing_names = {_clean(v["name"]).casefold() for v in registry.values()}

    src = _find_xlsx("additional", "barcode")
    print("Using:", os.path.basename(src))
    df = pd.read_excel(src)                       # header row 0
    name_c = next((c for c in df.columns
                   if "full name" in str(c).lower()
                   or str(c).lower().strip() == "name"), None)
    if name_c is None:
        raise SystemExit("No 'Full name' column found.")

    to_add = []
    for _, row in df.iterrows():
        raw = row.get(name_c, "")
        if pd.isna(raw):
            continue
        name = _clean(raw)
        if not name or name.lower() in ("nan", "nat"):
            continue
        if name.casefold() in existing_names:
            print(f"  skip (already in set): {name}")
            continue
        to_add.append(name)

    if not to_add:
        print("Nothing new to add.")
        return

    num = highest
    for name in to_add:
        num += 1
        line2 = f"{GROUP_LABEL} | {COMMITTEE}"
        safe = re.sub(r"[^A-Za-z0-9]+", "_", name).strip("_")
        path = os.path.join(OUTPUT_DIR, f"{num:03d}_{safe}.jpg")
        make_barcode_jpg(num, name, line2, path)
        registry[str(num)] = {"name": name, "group": GROUP_LABEL,
                              "committee": COMMITTEE, "position": ""}
        print(f"  added #{num}: {name}")

    with open(REGISTRY, "w", encoding="utf-8") as f:
        json.dump(registry, f, indent=2, ensure_ascii=False)
    print(f"\nDone — added {len(to_add)} barcode(s), "
          f"#{highest + 1}–#{num}. Total now {len(registry)}.")


if __name__ == "__main__":
    main()

The appender expects the original barcode registry and an Excel file named with “additional” and “barcode” in the Downloads folder. It should be run after the main barcode generator, not as a standalone generator.

4Making sure every meal was recorded

Generating barcodes was only half the job. At the event, we needed a way to scan someone's card and immediately know whether they'd already collected that particular meal. The last thing I wanted was for volunteers to wait for a spreadsheet to load while a long queue formed.

So I built a desktop barcode logger in Python using Tkinter. Before serving, the volunteer selects the meal and scans a participant's barcode. The program looks up that number in the delegate registry, checks its local record, and instantly shows either a green collected confirmation or a warning that the same meal has already been taken. An unknown barcode is rejected instead of silently being accepted.

The important design decision was to do the time-sensitive part locally. The program decides the scan result and updates the relevant row on screen first. After that, it saves the scan to a local SQLite database and uses a background process to send batches to Google Sheets. That means a temporary internet problem doesn't have to stop people from scanning cards. It also has an emergency log fallback if the database is unavailable.

Google Sheets is the shared source of truth: each machine rereads the scan log roughly every five seconds and rebuilds its board, while keeping recent local scans visible until they are synchronized. If an organizer corrects a record in the sheet, the board can reflect that change. The program also adapts the pace of its uploads when it hits API rate limits.

For the interface I used a virtualized Tkinter Canvas, drawing only the rows that are actually visible. Instead of creating thousands of separate widgets, it paints just the row that changed. It was one of the ways I tried to keep the experience responsive even with a large roster.

"""ISMUN barcode meal logger — local-first scans, Google-Sheet-as-truth board.

v4 changes (on top of the v3 latency/resilience rewrite)
--------------------------------------------------------
1. SHEET IS THE SOURCE OF TRUTH (requirements 2, 3, 4).
   The poller now does a FULL read of the Scan Log every POLL_INTERVAL_S
   (5 s) and REBUILDS the board state from it (reconcile), instead of
   incrementally merging only new rows. Consequences:
     - Rows DELETED in the sheet disappear from every machine's UI
       within one poll (an admin can now "un-serve" someone).
     - Rows ADDED or EDITED manually in the sheet (even without a ScanID)
       appear within one poll.
     - If the whole tab is wiped, the board clears AND the header/formulas
       are self-healed.
   Local scans that have not yet PROVABLY landed in the sheet are overlaid
   on top of the sheet truth so a fresh scan never flickers off the board:
     overlay = unsynced queue ∪ mem-pending ∪ recently-synced (≤30 s)
             ∪ recent local decides (≤30 s).
   Once a scan is durably in the sheet and past the overlay horizon, the
   sheet alone decides whether it exists.

2. VIRTUALIZED CANVAS BOARD (requirement 1).
   The old board was a grid of tk.Label widgets — 13 per person. At a few
   hundred people that is thousands of live widgets: slow to build, slow to
   scroll, and creating 13 widgets inside the scan handler was the source
   of the 10-20 ms latency outliers (requirement 5). The new VirtualBoard
   draws on a single Canvas and only ever materialises the ~40 visible
   rows, recycling them on scroll. Costs are now independent of roster
   size: a 10,000-person roster scrolls and paints the same as 10.
   The scan handler paints ONLY the scanned row before flushing pixels
   (the perceived-latency point), then completes the viewport.

   Your load test hit Google's 60-writes/min/user quota with three
   machines flushing every 2 s. The flusher now adapts: every 429 widens
   the per-machine flush gap (up to 15 s), every success decays it back
   toward 2 s. N machines converge on a fair share of the quota instead

# --------------------------------------------------------------------------- #
#  Durable local queue — SQLite, tier-1 persistence
# --------------------------------------------------------------------------- #
class DurableQueue:
    durable = True

    def __init__(self, path: str = DB_FILE):
        self.path = path
        self._local = threading.local()
        try:
            self._init_schema()
        except sqlite3.DatabaseError:
            logger.critical("scan DB corrupt/unusable — quarantining and "
                            "recreating", exc_info=True)
            self._quarantine()
            self._init_schema()

    def _quarantine(self) -> None:
        self._drop_thread_conn()
        for suffix in ("", "-wal", "-shm"):
            src = self.path + suffix
            if os.path.exists(src):
                dst = f"{src}.corrupt-{int(time.time())}"
                try:
                    os.replace(src, dst)
                    logger.critical("quarantined %s -> %s", src, dst)
                except OSError:
                    logger.critical("could not quarantine %s", src,
                                    exc_info=True)

    def _drop_thread_conn(self) -> None:
        con = getattr(self._local, "con", None)
        if con is not None:
            try:
                con.close()
            except sqlite3.Error:
                pass
            self._local.con = None

    def _init_schema(self) -> None:
        con = self._conn()
        ok = con.execute("PRAGMA quick_check").fetchone()[0]
        if ok != "ok":
            raise sqlite3.DatabaseError(f"quick_check: {ok}")
        con.execute("""
            CREATE TABLE IF NOT EXISTS scans (
                seq     INTEGER PRIMARY KEY AUTOINCREMENT,
                scan_id TEXT UNIQUE NOT NULL,
                ts      TEXT NOT NULL,
                pid     TEXT NOT NULL,
                name    TEXT,
                meal    TEXT NOT NULL,
                status  TEXT NOT NULL,
                machine TEXT NOT NULL,
                synced  INTEGER NOT NULL DEFAULT 0
            )""")
        con.execute("CREATE INDEX IF NOT EXISTS idx_unsynced "
                    "ON scans(synced) WHERE synced = 0")
        con.commit()

    def _conn(self) -> sqlite3.Connection:
        con = getattr(self._local, "con", None)
        if con is None:
            con = sqlite3.connect(self.path)
            con.execute("PRAGMA journal_mode=WAL")
            con.execute("PRAGMA synchronous=FULL")   # fsync per commit —
            con.execute("PRAGMA busy_timeout=5000")  # off the perceived path
            self._local.con = con
        return con

    def enqueue(self, rec: dict[str, Any]) -> None:
        con = self._conn()
        con.execute(
            "INSERT INTO scans(scan_id, ts, pid, name, meal, status, machine)"
            " VALUES (?,?,?,?,?,?,?)",
            (rec["scan_id"], rec["ts"], rec["pid"], rec["name"],
             rec["meal"], rec["status"], rec["machine"]))
        con.commit()

    def insert_ignore(self, rec: dict[str, Any]) -> bool:
        con = self._conn()
        cur = con.execute(
            "INSERT OR IGNORE INTO "
            "scans(scan_id, ts, pid, name, meal, status, machine)"
            " VALUES (?,?,?,?,?,?,?)",
            (rec["scan_id"], rec["ts"], rec["pid"], rec.get("name", ""),
             rec["meal"], rec["status"], rec["machine"]))
        con.commit()
        return cur.rowcount > 0

    def pending(self, limit: int = MAX_BATCH) -> list[dict[str, Any]]:
        cur = self._conn().execute(
            "SELECT scan_id, ts, pid, name, meal, status, machine"
            " FROM scans WHERE synced=0 ORDER BY seq LIMIT ?", (limit,))
        cols = ("scan_id", "ts", "pid", "name", "meal", "status", "machine")
        return [dict(zip(cols, row)) for row in cur.fetchall()]

    def mark_synced(self, scan_ids: list[str]) -> None:
        if not scan_ids:
            return
        con = self._conn()
        qmarks = ",".join("?" * len(scan_ids))
        con.execute(f"UPDATE scans SET synced=1 WHERE scan_id IN ({qmarks})",
                    scan_ids)
        con.commit()

    def pending_count(self) -> int:
        cur = self._conn().execute("SELECT COUNT(*) FROM scans WHERE synced=0")
        return int(cur.fetchone()[0])

    def all_rows(self) -> list[dict[str, Any]]:
        cur = self._conn().execute(
            "SELECT scan_id, ts, pid, name, meal, status, machine, synced"
            " FROM scans ORDER BY seq")
        cols = ("scan_id", "ts", "pid", "name", "meal",
                "status", "machine", "synced")
        return [dict(zip(cols, row)) for row in cur.fetchall()]


class MemoryQueue:
    """Last-resort stand-in when SQLite cannot be initialised at all.
    Same interface, NON-durable — the UI shows a permanent red banner."""

    durable = False

    def __init__(self) -> None:
        self._rows: list[dict[str, Any]] = []
        self._ids: set[str] = set()
        self._lock = threading.Lock()

    def enqueue(self, rec: dict[str, Any]) -> None:
        with self._lock:
            self._rows.append({**rec, "synced": 0})
            self._ids.add(rec["scan_id"])

    def insert_ignore(self, rec: dict[str, Any]) -> bool:
        with self._lock:
            if rec["scan_id"] in self._ids:
                return False
        self.enqueue(rec)
        return True

    def pending(self, limit: int = MAX_BATCH) -> list[dict[str, Any]]:
        with self._lock:
            out = [dict(r) for r in self._rows if not r["synced"]][:limit]
        for r in out:
            r.pop("synced", None)
        return out

    def mark_synced(self, scan_ids: list[str]) -> None:
        wanted = set(scan_ids)
        with self._lock:
            for r in self._rows:
                if r["scan_id"] in wanted:
                    r["synced"] = 1

    def pending_count(self) -> int:
        with self._lock:
            return sum(1 for r in self._rows if not r["synced"])

    def all_rows(self) -> list[dict[str, Any]]:
        with self._lock:
            return [dict(r) for r in self._rows]

# Board meal columns start at F (0-based index 5) and run len(MEALS) wide.
FIRST_MEAL_COL_IDX = 5                                   # A=0 … F=5
MEAL_COL_LETTERS = [chr(ord("A") + FIRST_MEAL_COL_IDX + i)
                    for i in range(len(MEALS))]          # F..Q for 12 meals
LAST_MEAL_COL = MEAL_COL_LETTERS[-1]
    
def _meal_formula(col_letter: str) -> str:
    L = "'" + LOG_SHEET + "'"
    return (
        '=ARRAYFORMULA(IF($A2:$A="","",'
        'IFERROR(REGEXEXTRACT(VLOOKUP($A2:$A&"§"&' + col_letter + '$1,'
        'FILTER({' + L + '!$B$2:$B&"§"&' + L + '!$D$2:$D,'
        + L + '!$A$2:$A},'
        + L + '!$E$2:$E="COLLECTED"),2,FALSE),'
        '"\\d\\d:\\d\\d:\\d\\d"),"")))'
    )

class SheetClient:
    def __init__(self) -> None:
        if not os.path.exists(SERVICE_ACCOUNT_FILE):
            raise RuntimeError(
                f"'{SERVICE_ACCOUNT_FILE}' not found. Set SCANNER_SA or fix "
                "the path at the top of this file.")
        if not SPREADSHEET_ID.strip():
            raise RuntimeError(
                "SPREADSHEET_ID is blank. Create a sheet, share it with the "
                "service account's client_email as Editor, and paste its ID.")
        self._creds = service_account.Credentials.from_service_account_file(
            SERVICE_ACCOUNT_FILE, scopes=SCOPES)
        self._local = threading.local()
        self.sid = SPREADSHEET_ID.strip()
        self._setup()

    @property
    def svc(self):  # type: ignore[no-untyped-def]
        s = getattr(self._local, "svc", None)
        if s is None:
            s = build("sheets", "v4", credentials=self._creds,
                      cache_discovery=False)
            self._local.svc = s
        return s

    def _setup(self) -> None:
        try:
            meta = self.svc.spreadsheets().get(spreadsheetId=self.sid).execute()
        except HttpError as e:
            raise RuntimeError(
                f"Could not open sheet '{self.sid}'. Check the ID and that "
                "the sheet is shared (Editor) with the service account's "
                f"client_email. Original error: {e}") from e

        def titles_of(m: dict[str, Any]) -> dict[str, int]:
            return {s["properties"]["title"]: s["properties"]["sheetId"]
                    for s in m.get("sheets", [])}

        titles = titles_of(meta)
        reqs = [{"addSheet": {"properties": {"title": t}}}
                for t in (LOG_SHEET, BOARD_SHEET) if t not in titles]
        if reqs:
            self.svc.spreadsheets().batchUpdate(
                spreadsheetId=self.sid, body={"requests": reqs}).execute()
            meta = self.svc.spreadsheets().get(
                spreadsheetId=self.sid).execute()
            titles = titles_of(meta)
        if "Sheet1" in titles and len(titles) > 1:
            self.svc.spreadsheets().batchUpdate(
                spreadsheetId=self.sid,
                body={"requests": [
                    {"deleteSheet": {"sheetId": titles["Sheet1"]}}]}).execute()
            del titles["Sheet1"]

        self.svc.spreadsheets().values().update(
            spreadsheetId=self.sid, range=f"{LOG_SHEET}!A1",
            valueInputOption="RAW", body={"values": [LOG_HEADER]}).execute()
        self.svc.spreadsheets().values().update(
            spreadsheetId=self.sid, range=f"{BOARD_SHEET}!A1",
            valueInputOption="RAW", body={"values": [COLUMNS]}).execute()

        self._ensure_board_machinery(titles[BOARD_SHEET])

    def re_setup(self) -> None:
        """Recovery hook: tab deleted/renamed/wiped mid-event."""
        self._setup()

    def verify_and_heal(self) -> bool:
        """Cheap header probe; full re-setup if the Scan Log was wiped.
        Returns True if healing ran."""
        res = self.svc.spreadsheets().values().get(
            spreadsheetId=self.sid, range=f"{LOG_SHEET}!A1:A1").execute()
        cell = (res.get("values") or [[""]])[0][0] if res.get("values") else ""
        if cell != LOG_HEADER[0]:
            logger.warning("Scan Log header missing — re-initialising "
                           "workbook (was the sheet erased?)")
            self._setup()
            return True
        return False

    def _ensure_board_machinery(self, board_sheet_id: int) -> None:
        # Probe the WHOLE meal-formula strip, not just F2. A sheet built by
        # the old 9-meal build has F2 populated but O2:Q2 empty — probing F2

        res = self.svc.spreadsheets().values().get(
            spreadsheetId=self.sid,
            range=f"{BOARD_SHEET}!F2:{LAST_MEAL_COL}2",
            valueRenderOption="FORMULA").execute()
        vals = res.get("values") or []
        row = list(vals[0]) if vals else []
        row += [""] * (len(MEALS) - len(row))
        if all(str(c).startswith("=ARRAYFORMULA") for c in row):
            return

        self.svc.spreadsheets().values().update(
            spreadsheetId=self.sid,
            range=f"{BOARD_SHEET}!F2:{LAST_MEAL_COL}2",
            valueInputOption="USER_ENTERED",
            body={"values": [[_meal_formula(c)
                              for c in MEAL_COL_LETTERS]]}).execute()

        header_blue = {"red": 0.122, "green": 0.306, "blue": 0.470}
        green_fill  = {"red": 0.776, "green": 0.937, "blue": 0.808}
        requests = [
            {
                "repeatCell": {
                    "range": {"sheetId": board_sheet_id,
                              "startRowIndex": 0, "endRowIndex": 1,
                              "startColumnIndex": 0,
                              "endColumnIndex": len(COLUMNS)},
                    "cell": {"userEnteredFormat": {
                        "backgroundColor": header_blue,
                        "horizontalAlignment": "CENTER",
                        "verticalAlignment": "MIDDLE",
                        "textFormat": {"bold": True, "foregroundColor":
                                       {"red": 1, "green": 1, "blue": 1}}}},
                    "fields": "userEnteredFormat(backgroundColor,"
                              "horizontalAlignment,verticalAlignment,"
                              "textFormat)",
                }
            },
            {
                "updateSheetProperties": {
                    "properties": {"sheetId": board_sheet_id,
                                   "gridProperties": {"frozenRowCount": 1,
                                                      "frozenColumnCount": 2}},
                    "fields": "gridProperties(frozenRowCount,"
                              "frozenColumnCount)",
                }
            },
            {
                "addConditionalFormatRule": {
                    "index": 0,
                    "rule": {
                        "ranges": [{"sheetId": board_sheet_id,
                                    "startRowIndex": 1, "endRowIndex": 10000,
                                    "startColumnIndex": FIRST_MEAL_COL_IDX,
                                    "endColumnIndex": len(COLUMNS)}],
                        "booleanRule": {
                            "condition": {
                                "type": "CUSTOM_FORMULA",
                                "values": [{"userEnteredValue": '=F2<>""'}]},
                            "format": {"backgroundColor": green_fill}},
                    },
                }
            },
        ]
        self.svc.spreadsheets().batchUpdate(
            spreadsheetId=self.sid, body={"requests": requests}).execute()

# --------------------------------------------------------------------------- #
#  Client manager — builds SheetClient in the background with retry
# --------------------------------------------------------------------------- #
class SheetClientManager(threading.Thread):
    def __init__(self, ui_events: "queue.Queue[tuple[str, Any]]",
                 factory: Callable[[], Any] = None) -> None:  # type: ignore
        super().__init__(daemon=True, name="client-init")
        self.ui = ui_events
        self._factory = factory or SheetClient
        self._client: Any = None
        self._lock = threading.Lock()

    def get(self) -> Any:
        with self._lock:
            return self._client

    def run(self) -> None:
        backoff = BACKOFF_BASE_S
        while True:
            try:
                c = self._factory()
            except Exception as e:
                logger.error("Sheets client init failed [%s]: %r — retrying "
                             "in %.0fs (scans keep queueing locally)",
                             classify_error(e), e, backoff)
                self.ui.put(("offline-init", None))
                time.sleep(backoff + random.uniform(0, backoff))
                backoff = min(BACKOFF_CAP_S, backoff * 2)
                continue
            with self._lock:
                self._client = c
            logger.info("Sheets client ready")
            self.ui.put(("online", None))
            return


# --------------------------------------------------------------------------- #
#  Store — local state + persistence fallback chain + sheet reconciliation
# --------------------------------------------------------------------------- #
class Store:
    def __init__(self, dq: DurableQueue | MemoryQueue,
                 machine_id: str = MACHINE_ID) -> None:
        self.dq = dq
        self.machine_id = machine_id
        self.lock = threading.RLock()
        self.registry = load_registry()
        self._num_index = build_numeric_index(self.registry)
        self._reg_mtime = _registry_mtime()
        self.state: dict[str, dict[str, datetime]] = {}
        self.mem_pending: deque[dict[str, Any]] = deque()   # tier-3 records
        # Overlay sources for reconcile: local activity that may not be
        # visible in a (possibly seconds-stale) sheet snapshot yet.
        self.recent_synced: deque[tuple[float, dict[str, Any]]] = deque()
        self.recent_decides: deque[tuple[float, dict[str, Any]]] = deque()
        self.flush_event = threading.Event()

    # ---- registry ---------------------------------------------------------- #
    def reload_registry_if_changed(self) -> bool:
        m = _registry_mtime()
        if m != self._reg_mtime:
            reg = load_registry()             # parse OUTSIDE the lock
            idx = build_numeric_index(reg)
            with self.lock:
                self.registry = reg
                self._num_index = idx
                self._reg_mtime = m
            logger.info("registry reloaded (%d ids)", len(reg))
            return True
        return False

    def resolve(self, code: str) -> str | None:
        """O(1): exact id hit, else unique-numeric-suffix hit."""
        code = code.strip()
        with self.lock:
            if code in self.registry:
                return code
            if code.isdigit():
                return self._num_index.get(int(code))
        return None

    def roster_ids(self) -> list[str]:
        def key(rid: str) -> int:
            m = _NUM_SUFFIX.search(rid)
            return int(m.group(1)) if m else 0
        with self.lock:
            return sorted(self.registry, key=key)

    def person_fields(self, pid: str) -> tuple[str, str, str, str]:
        rec = self.registry.get(pid, {})
        name = rec.get("name", "")
        aff = (rec.get("committee", "") or "").strip()
        group = (rec.get("group", "") or "").strip()
        if aff.upper() == "SEC":
            deleg, oc, desig = "SEC", "SEC", "SEC"
        elif aff.upper() in OC_TEAMS:
            deleg, oc, desig = "-", aff, (group or "-")
        else:
            deleg, oc, desig = (aff or "-"), "-", (group or "-")
        return name, deleg, oc, desig

    def board_roster_rows(self) -> list[list[str]]:
        return [[pid, *self.person_fields(pid)] for pid in self.roster_ids()]

    def collection_time(self, pid: str, meal: str) -> datetime | None:
        with self.lock:
            return self.state.get(pid, {}).get(meal)

    def known_pids(self) -> list[str]:
        """Roster order, then any extra pids present only in state."""
        roster = self.roster_ids()
        with self.lock:
            extras = sorted(set(self.state) - set(roster))
        return roster + extras

    def unsynced_count(self) -> int:
        try:
            n = self.dq.pending_count()
        except Exception:
            logger.exception("pending_count failed")
            n = 0
        with self.lock:
            return n + len(self.mem_pending)

    # ---- crash recovery: provisional local truth until the first poll ------- #
    def bootstrap_local(self) -> None:
        try:
            rows = self.dq.all_rows()
        except Exception:
            logger.exception("bootstrap: could not read local queue")
            rows = []
        with self.lock:
            for r in rows:
                if r["status"] != "COLLECTED":
                    continue
                try:
                    ts = datetime.strptime(r["ts"], TS_FMT)
                except (ValueError, TypeError):
                    continue
                slot = self.state.setdefault(r["pid"], {})
                if r["meal"] not in slot or ts < slot[r["meal"]]:
                    slot[r["meal"]] = ts

    # ---- PHASE 1: the perceived scan path — pure memory --------------------- #
    def scan_decide(self, pid: str, meal: str) -> dict[str, Any]:
        now = datetime.now()
        scan_id = f"{self.machine_id}-{uuid.uuid4().hex[:12]}"
        rec = {"scan_id": scan_id, "ts": now.strftime(TS_FMT),
               "pid": pid, "name": "", "meal": meal,
               "status": "", "machine": self.machine_id}
        with self.lock:
            name, _, _, _ = self.person_fields(pid)
            slot = self.state.setdefault(pid, {})
            if meal in slot:
                status, first = "ALREADY", slot[meal]
            else:
                slot[meal] = now
                status, first = "COLLECTED", now
            rec["name"], rec["status"] = name, status
            if status == "COLLECTED":
                # Overlay source: protects this scan from being wiped by a
                # reconcile built from a sheet snapshot that predates it.
                self.recent_decides.append((time.monotonic(), rec))
        return {"status": status, "first": first, "name": name, "rec": rec}

    # ---- PHASE 2: durability, three-tier fallback chain ---------------------- #
    def scan_persist(self, rec: dict[str, Any]) -> str:
        try:
            self.dq.enqueue(rec)
            return "sqlite"
        except Exception:
            logger.exception("SQLite enqueue failed — falling back "
                             "(scan_id=%s)", rec["scan_id"])
        with self.lock:
            self.mem_pending.append(rec)
        return "emergency" if emergency_append(rec) else "memory"

    def drop_mem_pending(self, scan_ids: set[str]) -> None:
        with self.lock:
            self.mem_pending = deque(
                r for r in self.mem_pending if r["scan_id"] not in scan_ids)

    def note_synced(self, recs: list[dict[str, Any]]) -> None:
        now = time.monotonic()
        with self.lock:
            self.recent_synced.extend((now, r) for r in recs)

    # ---- SHEET-AS-TRUTH RECONCILIATION -------------------------------------- #
    def reconcile(self, sheet_rows: list[list[str]]) -> set[tuple[str, str]]:
        """Rebuild board state from the sheet (truth), overlaid with local
        scans that may not be visible in this snapshot yet. Returns every
        (pid, meal) cell whose value changed — including cells that were
        CLEARED because their rows were deleted from the sheet."""
        new_state: dict[str, dict[str, datetime]] = {}
        seen_ids: set[str] = set()

        def merge(pid: str, meal: str, ts: datetime) -> None:
            slot = new_state.setdefault(pid, {})
            if meal not in slot or ts < slot[meal]:
                slot[meal] = ts                       # earliest-wins

        # 1) The sheet — authoritative. Manual rows (no ScanID) count too.
        for raw in sheet_rows:
            try:
                r = (list(raw) + [""] * 7)[:7]
                ts_s, pid, _name, meal, status, scan_id, _machine = (
                    str(x).strip() for x in r)
            except (TypeError, ValueError):
                logger.debug("malformed sheet row skipped: %.120r", raw)
                continue
            if status != "COLLECTED" or not pid or meal not in MEAL_SET:
                continue
            if scan_id:
                if scan_id in seen_ids:               # duplicate append —
                    continue                          # idempotent, skip
                seen_ids.add(scan_id)
            try:
                ts = datetime.strptime(ts_s, TS_FMT)
            except (ValueError, TypeError):
                logger.debug("bad timestamp in sheet row skipped: %.120r", raw)
                continue
            merge(pid, meal, ts)

        # 2) Overlay — local activity the snapshot may not include yet:
        #    unsynced queue, mem-pending, recently synced, recent decides.
        overlay: list[dict[str, Any]] = []
        try:
            overlay.extend(self.dq.pending(1_000_000_000))
        except Exception:
            logger.exception("reconcile: local queue read failed")
        horizon = time.monotonic() - OVERLAY_HORIZON_S
        with self.lock:
            while self.recent_synced and self.recent_synced[0][0] < horizon:
                self.recent_synced.popleft()
            while self.recent_decides and self.recent_decides[0][0] < horizon:
                self.recent_decides.popleft()
            overlay.extend(self.mem_pending)
            overlay.extend(r for _, r in self.recent_synced)
            overlay.extend(r for _, r in self.recent_decides)

        for r in overlay:
            if r.get("status") != "COLLECTED":
                continue
            pid, meal = r.get("pid", ""), r.get("meal", "")
            if not pid or meal not in MEAL_SET:
                continue
            try:
                ts = datetime.strptime(r["ts"], TS_FMT)
            except (KeyError, ValueError, TypeError):
                continue
            merge(pid, meal, ts)

        # 3) Swap + diff under the lock (µs–ms; a concurrent scan_decide's
        #    mutation is preserved because it is in recent_decides above,
        #    or happens after the swap against the new state).
        changed: set[tuple[str, str]] = set()
        with self.lock:
            old = self.state
            for pid in old.keys() | new_state.keys():
                om, nm = old.get(pid, {}), new_state.get(pid, {})
                for meal in om.keys() | nm.keys():
                    if om.get(meal) != nm.get(meal):
                        changed.add((pid, meal))
            self.state = new_state
        return changed


# --------------------------------------------------------------------------- #
#  Flusher — the only writer. Adaptive pacing against the shared write quota.
# --------------------------------------------------------------------------- #
class Flusher(threading.Thread):
    def __init__(self, store: Store, mgr: SheetClientManager,
                 ui_events: "queue.Queue[tuple[str, Any]]") -> None:
        super().__init__(daemon=True, name="flusher")
        self.store, self.mgr, self.ui = store, mgr, ui_events
        self._backoff = BACKOFF_BASE_S
        self._gap = MIN_FLUSH_GAP_S           # adaptive: grows on 429s
        self._last_flush = 0.0

    def run(self) -> None:
        while True:
            try:
                self._cycle()
            except Exception:
                logger.exception("flusher: unexpected error — recovering")
                time.sleep(5.0)

    def _cycle(self) -> None:
        self.store.flush_event.wait(timeout=2.0)
        self.store.flush_event.clear()

        client = self.mgr.get()
        if client is None:
            return
        if self.store.unsynced_count() == 0:
            return

        time.sleep(FLUSH_WINDOW_S)
        gap = self._gap - (time.monotonic() - self._last_flush)
        if gap > 0:
            time.sleep(gap)

        try:
            batch = self.store.dq.pending(MAX_BATCH)
        except Exception:
            logger.exception("flusher: local queue read failed")
            batch = []
        with self.store.lock:
            room = MAX_BATCH - len(batch)
            mem = list(self.store.mem_pending)[:max(0, room)]
        if not batch and not mem:
            return

        rows = batch + mem
        values = [[b["ts"], b["pid"], b["name"], b["meal"],
                   b["status"], b["scan_id"], b["machine"]] for b in rows]
        try:
            client.append_log_rows(values)
        except Exception as e:
            kind = classify_error(e)
            if kind == "rate-limit":
                # Widen this machine's share of the 60 writes/min/user
                # quota; success decays it back. N machines converge on a
                # fair cadence instead of retry-storming (your load test).
                self._gap = min(MAX_FLUSH_GAP_S, self._gap * 1.5 + 1.0)
            if kind == "auth":
                logger.critical("flusher: 401/403 — check that the service "
                                "account still has Editor access")
            logger.warning("flusher: append failed [%s]: %r — backing off "
                           "%.1fs (gap now %.1fs), queue is safe locally",
                           kind, e, self._backoff, self._gap)
            self.ui.put(("offline", self.store.unsynced_count()))
            time.sleep(self._backoff + random.uniform(0, self._backoff))
            self._backoff = min(BACKOFF_CAP_S, self._backoff * 2)
            self.store.flush_event.set()
            return

        try:
            self.store.dq.mark_synced([b["scan_id"] for b in batch])
        except Exception:
            logger.exception("flusher: mark_synced failed — duplicate "
                             "appends possible (idempotent, board unaffected)")
        self.store.drop_mem_pending({b["scan_id"] for b in mem})
        self.store.note_synced(rows)          # overlay: no flicker vs stale
                                              # sheet snapshots
        self._last_flush = time.monotonic()
        self._backoff = BACKOFF_BASE_S
        self._gap = max(MIN_FLUSH_GAP_S, self._gap * 0.9)
        n = self.store.unsynced_count()
        self.ui.put(("synced", n))
        if n:
            self.store.flush_event.set()


# --------------------------------------------------------------------------- #
#  Poller — full read + reconcile every POLL_INTERVAL_S. The sheet is truth.
# --------------------------------------------------------------------------- #
class Poller(threading.Thread):
    def __init__(self, store: Store, mgr: SheetClientManager,
                 ui_events: "queue.Queue[tuple[str, Any]]") -> None:
        super().__init__(daemon=True, name="poller")
        self.store, self.mgr, self.ui = store, mgr, ui_events
        self._delay = 0.5                     # first poll: fast
        self._roster_pushed = False
        self._had_rows = False

    def run(self) -> None:
        while True:
            time.sleep(self._delay)
            try:
                self._delay = self._cycle()
            except Exception as e:
                kind = classify_error(e)
                self._delay = min(60.0, max(self._delay, 1.0) * 2)
                logger.warning("poller: cycle failed [%s]: %r — next poll in "
                               "%.0fs", kind, e, self._delay)
                if isinstance(e, HttpError):
                    status = getattr(getattr(e, "resp", None), "status", None)
                    if status in (400, 404):
                        self._try_re_setup()

    def _try_re_setup(self) -> None:
        client = self.mgr.get()
        if client is None:
            return
        try:
            client.re_setup()
            logger.warning("poller: workbook re-initialised after 400/404")
        except Exception:
            logger.exception("poller: re-setup failed")

    def _cycle(self) -> float:
        client = self.mgr.get()
        if client is None:
            return 1.0

        if not self._roster_pushed:
            client.push_roster(self.store.board_roster_rows())
            self._roster_pushed = True

        if self.store.reload_registry_if_changed():
            client.push_roster(self.store.board_roster_rows())
            self.ui.put(("roster", None))

        rows = client.read_log_all()          # FULL read: sheet is truth
        if not rows and self._had_rows:
            # The log had data and now has none: either an admin wiped it
            # (legitimate — the board must clear) or the tab was damaged.
            # Heal the header/formulas either way; the empty truth stands.
            try:
                if client.verify_and_heal():
                    self.ui.put(("healed", None))
            except Exception:
                logger.exception("poller: heal probe failed")
        self._had_rows = bool(rows)

        changed = self.store.reconcile(rows)
        if changed:
            self.ui.put(("cells", changed))
        return POLL_INTERVAL_S

#  exist as canvas items; they are recycled on scroll. All costs are O(visible)

        self._cache: list[list[tuple[str, str, str] | None]] = []
        self._flash: dict[str, str] = {}       # pid -> flash colour

                                highlightthickness=0)

                                  command=self._on_vbar)

                                  command=self._xview)

        self.body.grid(row=1, column=0, sticky="nsew")
        self.vbar.grid(row=1, column=1, sticky="ns")
        self.hbar.grid(row=2, column=0, sticky="ew")
        self.grid_rowconfigure(1, weight=1)
        self.grid_columnconfigure(0, weight=1)

        self.header.configure(scrollregion=(0, 0, self._total_w, HEADER_H))
        self.body.configure(scrollregion=(0, 0, self._total_w, 1),

        x = 0
        for c, title in enumerate(COLUMNS):
            w = COL_PX[c]
            self.header.create_rectangle(x, 0, x + w, HEADER_H,
                                         fill=GRID, outline=BG)
            anchor, tx = (("w", x + 8) if c in (0, 1, 2)

                                    anchor=anchor,
                                    font=("Segoe UI Semibold", 10))

        for w in (self.body, self.header):
            w.bind("<MouseWheel>", self._on_wheel)
            w.bind("<Button-4>", lambda e: self._wheel_steps(-3))   # X11
            w.bind("<Button-5>", lambda e: self._wheel_steps(3))

    # ---- geometry / pooling ------------------------------------------------ #
    def _visible_rows(self) -> int:
        return max(1, self.body.winfo_height() // ROW_H)

    def _on_resize(self, _e: object = None) -> None:
        need = self._visible_rows() + 1
        while self._nslots < need:
            s = self._nslots
            y0 = s * ROW_H
            row: list[tuple[int, int]] = []
            x = 0
            for c in range(len(COLUMNS)):
                w = COL_PX[c]
                r = self.body.create_rectangle(x, y0, x + w, y0 + ROW_H,
                                               fill=BG, outline=GRID)
                anchor, tx = (("w", x + 8) if c in (0, 1, 2)
                              else ("center", x + w / 2))
                t = self.body.create_text(tx, y0 + ROW_H / 2, text="",
                                          fill=TEXT, anchor=anchor,
                                          font=("Segoe UI", 10))
                row.append((r, t))
                x += w
            self._slots.append(row)
            self._cache.append([None] * len(COLUMNS))
            self._nslots += 1
        self.render()

    # ---- scrolling ---------------------------------------------------------- #
    def _clamp_first(self) -> None:
        self.first = max(0, min(self.first,
                                max(0, len(self.order) - self._visible_rows())))

    def _on_vbar(self, *args: str) -> None:
        n = len(self.order)
        if not n:
            return
        if args[0] == "moveto":
            self.first = int(float(args[1]) * n)
        elif args[0] == "scroll":
            step = int(args[1])
            if args[2] == "pages":
                step *= self._visible_rows()
            self.first += step
        self._clamp_first()
        self.render()

    def _wheel_steps(self, rows: int) -> None:
        self.first += rows
        self._clamp_first()
        self.render()

    def _on_wheel(self, e: "tk.Event[Any]") -> None:
        if e.delta:
            self._wheel_steps(-3 if e.delta > 0 else 3)

    def _xview(self, *args: str) -> None:
        self.body.xview(*args)

    def _on_xset(self, lo: str, hi: str) -> None:
        self.hbar.set(lo, hi)
        self.header.xview_moveto(lo)          # keep header aligned

    def _update_vbar(self) -> None:
        n = len(self.order)
        vis = self._visible_rows()
        if n <= vis:
            self.vbar.set(0.0, 1.0)
        else:
            self.vbar.set(self.first / n, (self.first + vis) / n)

    # ---- data model ---------------------------------------------------------- #
    def set_order(self, pids: list[str]) -> None:
        self.order = list(pids)
        self.pos = {pid: i for i, pid in enumerate(self.order)}
        self._clamp_first()
        self.render()

    def ensure_pid(self, pid: str) -> None:
        if pid not in self.pos:
            self.pos[pid] = len(self.order)
            self.order.append(pid)

    # ---- rendering ------------------------------------------------------------ #
    def _row_content(self, pid: str | None) -> list[tuple[str, str, str]]:
        if pid is None:
            return [("", BG, TEXT)] * len(COLUMNS)
        flash = self._flash.get(pid)
        name, deleg, oc, desig = self.store.person_fields(pid)
        out: list[tuple[str, str, str]] = [
            (pid, CARD, TEXT), (name, CARD, TEXT),
            (deleg, CARD, TEXT), (oc, CARD, TEXT), (desig, CARD, TEXT)]
        for meal in MEALS:
            t = self.store.collection_time(pid, meal)
            if t:
                out.append((t.strftime("%H:%M"), CELL_GREEN, GREEN_FG))
            else:
                out.append(("", CARD, TEXT))
        if flash:
            out = [(txt, flash, FLASH_FG) for txt, _, _ in out]
        return out

    def _render_slot(self, s: int) -> None:
        i = self.first + s
        pid = self.order[i] if 0 <= i < len(self.order) else None
        content = self._row_content(pid)
        cache = self._cache[s]
        cfg = self.body.itemconfigure
        for c, cell in enumerate(content):
            if cache[c] != cell:
                rect, text = self._slots[s][c]
                cfg(rect, fill=cell[1])
                cfg(text, text=cell[0], fill=cell[2])
                cache[c] = cell

    def render(self) -> None:
        self._update_vbar()
        for s in range(self._nslots):
            self._render_slot(s)

    def render_pid(self, pid: str) -> None:
        """Repaint just one row if visible — the perceived-latency path."""
        i = self.pos.get(pid)
        if i is None:
            return
        s = i - self.first
        if 0 <= s < self._nslots:
            self._render_slot(s)

    # ---- scan feedback ----------------------------------------------------------- #
    def scroll_to(self, pid: str) -> bool:
        """Bring pid into view. Returns True if the viewport moved."""
        i = self.pos.get(pid)
        if i is None:
            return False
        vis = self._visible_rows()
        if self.first <= i < self.first + vis:
            return False
        self.first = max(0, min(i - vis // 2, max(0, len(self.order) - vis)))
        return True

    def flash(self, pid: str, colour: str,
              after: Callable[[int, Callable[[], None]], object]) -> None:
        self._flash[pid] = colour
        after(900, lambda: self._end_flash(pid))

    def _end_flash(self, pid: str) -> None:
        self._flash.pop(pid, None)
        self.render_pid(pid)

5Testing how fast and reliable it was

I didn't want to call the system fast just because it felt fast on my laptop. I wanted to know what would happen with hundreds of people, thousands of previous scans, several scanners working together, or an internet connection that suddenly stopped working. That's why I wrote a separate benchmarking program.

The benchmark imports the actual barcode logger and drives its normal code paths using fake participants and meal records. It runs inside a temporary folder, substitutes a fake Google Sheets client, and checks whether the real application files stayed untouched. That lets me stress-test the program without modifying the event's attendance or meal records.

First, it times the smallest operations: identifying a barcode, checking whether a meal has already been collected, committing a scan to SQLite, and counting scans still waiting to upload. It measures these with Python's high-resolution performance clock and reports the median, 90th percentile, 99th percentile, and worst observed time. The 99th percentile is particularly useful because a system can feel slow due to occasional delays even when its average is excellent.

Then it makes the test harder. It grows the database and simulated spreadsheet, checks how long reconciliation holds shared locks, and puts multiple operations under concurrent load. When a graphical display is available, it also benchmarks the scrolling board, the actual scan-to-screen-update path, and UI responsiveness. Other phases simulate upload delays, server errors, disconnected operation, recovery, and longer-running event traffic.

One subtle distinction is that the benchmark measures perceived latency separately from durable storage and cloud synchronization. Perceived latency ends when the scan result is painted on the screen; the SQLite commit and Google Sheets upload happen afterward. So a tiny time for the in-memory decision isn't the same as the time for a whole scan to be safely saved and synchronized.

About the results: this page explains the benchmark methodology, not a new verified benchmark run. The source includes pass/warn/fail thresholds, but I haven't included a performance number here without an actual test report and the machine specifications. Any sub-millisecond or nanosecond result should specify exactly what operation it measures.

class FakeSheet:
    """Stand-in for SheetClient. Implements the full pipeline API.
    Optional simulated network latency and 429 injection."""

    def __init__(self, latency_s: float = 0.0, fail_every: int = 0):
        self.rows: list[list[str]] = []
        self.roster: list[list[str]] = []
        self.latency = latency_s
        self.fail_every = fail_every
        self.appends = 0
        self.reads = 0
        self.roster_pushes = 0
        self.heals = 0
        self.bytes_read = 0
        self._lock = threading.Lock()

    def _net(self):
        if self.latency:
            time.sleep(self.latency)

    def append_log_rows(self, values):
        self._net()
        with self._lock:
            self.appends += 1
            if self.fail_every and self.appends % self.fail_every == 0:
                raise RuntimeError("simulated 429 / transport error")
            self.rows.extend([list(map(str, v)) for v in values])

    def read_log_all(self):
        self._net()
        with self._lock:
            self.reads += 1
            snap = [list(r) for r in self.rows]
        self.bytes_read += sum(len(c) for r in snap for c in r)
        return snap

    def push_roster(self, rows):
        self._net()
        self.roster_pushes += 1
        self.roster = [list(r) for r in rows]

    def re_setup(self):
        self._net()

    def verify_and_heal(self):
        self._net()
        self.heals += 1
        return False

# --------------------------------------------------------------------------- #
#  4. Instrumented lock — measures contention inside Store
# --------------------------------------------------------------------------- #
class InstrumentedRLock:
    """Drop-in for threading.RLock that records acquire-wait and hold times.
    Reentrancy-aware: only the outermost acquire/release pair is timed."""

    def __init__(self) -> None:
        self._lock = threading.RLock()
        self._tl = threading.local()
        self.waits_ns: list[int] = []
        self.holds_ns: list[int] = []
        self.enabled = False
        self._guard = threading.Lock()

    def acquire(self, blocking=True, timeout=-1):
        depth = getattr(self._tl, "depth", 0)
        if depth == 0 and self.enabled:
            t0 = time.perf_counter_ns()
            got = self._lock.acquire(blocking, timeout)
            t1 = time.perf_counter_ns()
            if got:
                with self._guard:
                    self.waits_ns.append(t1 - t0)
                self._tl.t_in = t1
        else:
            got = self._lock.acquire(blocking, timeout)
        if got:
            self._tl.depth = depth + 1
        return got

    def release(self):
        depth = getattr(self._tl, "depth", 1)
        self._tl.depth = depth - 1
        if depth - 1 == 0 and self.enabled:
            t_in = getattr(self._tl, "t_in", None)
            if t_in is not None:
                with self._guard:
                    self.holds_ns.append(time.perf_counter_ns() - t_in)
                self._tl.t_in = None
        self._lock.release()

    __enter__ = acquire

    def __exit__(self, *exc):
        self.release()
        return False

    def reset(self):
        with self._guard:
            self.waits_ns.clear()
            self.holds_ns.clear()

def pct(xs: list[float], q: float) -> float:
    if not xs:
        return float("nan")
    s = sorted(xs)
    k = (len(s) - 1) * q
    lo, hi = int(k), min(int(k) + 1, len(s) - 1)
    return s[lo] + (s[hi] - s[lo]) * (k - lo)


class Stat:
    def __init__(self, name: str, samples_ms: list[float], unit: str = "ms"):
        self.name = name
        self.s = samples_ms
        self.unit = unit

    @property
    def n(self):
        return len(self.s)

    def row(self) -> str:
        if not self.s:
            return f"{self.name:<44} {'(no samples)':>30}"
        return (f"{self.name:<44} n={self.n:<6} "
                f"p50={pct(self.s, .50):8.3f} p90={pct(self.s, .90):8.3f} "
                f"p99={pct(self.s, .99):8.3f} max={max(self.s):8.3f}")


HDR = (f"{'measurement':<44} {'count':<8} "
       f"{'p50 ms':>8} {'p90 ms':>8} {'p99 ms':>8} {'max ms':>8}")


def table(stats: list[Stat]) -> None:
    say(HDR)
    say("-" * 92)
    for st in stats:
        say(st.row())

def make_sheet_rows(n_people: int, n_rows: int) -> list[list[str]]:
    """Realistic Scan Log rows: [ts, pid, name, meal, status, scanid, mach]."""
    base = datetime(2026, 8, 1, 8, 0, 0)
    rows = []
    meals = bc.MEALS
    for i in range(n_rows):
        pid = f"ISMUN-{(i % n_people) + 1:04d}"
        meal = meals[(i // max(1, n_people)) % len(meals)]
        ts = (base + timedelta(seconds=i * 3)).strftime(bc.TS_FMT)
        rows.append([ts, pid, f"Delegate Number {(i % n_people) + 1}", meal,
                     "COLLECTED", f"BENCH-{i:08d}", "BENCH-STATION"])
    return rows


def fresh_store(n_people: int, db_path: str | None = None):
    """A brand-new Store on a brand-new DB, with an instrumented lock."""
    write_registry(n_people)
    path = db_path or os.path.join(SANDBOX, f"db_{n_people}_{time.time_ns()}.db")
    dq = bc.DurableQueue(path)
    store = bc.Store(dq)
    store.lock = InstrumentedRLock()
    return store, dq, path

#  PHASE 1 — micro-benchmarks on the hot scan path
# --------------------------------------------------------------------------- #
def phase1_micro() -> None:
    rule("PHASE 1 — hot-path micro-benchmarks (no UI, no network)")
    say("Every operation the scan handler performs, timed in isolation.\n")

    N_PEOPLE = 200 if QUICK else 1000
    ITER = 2000 if QUICK else 8000
    PERSIST_ITER = 400 if QUICK else 1500

    store, dq, _ = fresh_store(N_PEOPLE)
    pids = store.roster_ids()

    # --- resolve(): exact id hit ---
    s_exact = []
    for i in range(ITER):
        code = pids[i % len(pids)]
        t = time.perf_counter_ns()
        store.resolve(code)
        s_exact.append((time.perf_counter_ns() - t) / 1e6)

    # --- resolve(): numeric-suffix hit (what the scanner gun sends) ---
    s_num = []
    for i in range(ITER):
        code = str((i % N_PEOPLE) + 1)
        t = time.perf_counter_ns()
        store.resolve(code)
        s_num.append((time.perf_counter_ns() - t) / 1e6)

    # --- resolve(): miss (unknown barcode -> rejection path) ---
    s_miss = []
    for i in range(ITER):
        t = time.perf_counter_ns()
        store.resolve(f"GARBAGE-{i}")
        s_miss.append((time.perf_counter_ns() - t) / 1e6)

    # --- person_fields() ---
    s_pf = []
    for i in range(ITER):
        pid = pids[i % len(pids)]
        t = time.perf_counter_ns()
        store.person_fields(pid)
        s_pf.append((time.perf_counter_ns() - t) / 1e6)

    # --- scan_decide(): the in-memory decision, COLLECTED path ---
    s_dec_new, s_dec_dup = [], []
    combos = [(p, m) for m in bc.MEALS for p in pids]
    for i, (pid, meal) in enumerate(combos[:ITER]):
        t = time.perf_counter_ns()
        store.scan_decide(pid, meal)
        s_dec_new.append((time.perf_counter_ns() - t) / 1e6)

    # --- scan_decide(): the ALREADY path (repeat scan) ---
    for i, (pid, meal) in enumerate(combos[:min(ITER, 2000)]):
        t = time.perf_counter_ns()
        store.scan_decide(pid, meal)
        s_dec_dup.append((time.perf_counter_ns() - t) / 1e6)

    # --- scan_persist(): SQLite INSERT + commit with synchronous=FULL ---
    s_persist = []
    for i in range(PERSIST_ITER):
        rec = {"scan_id": f"P-{i:09d}", "ts": datetime.now().strftime(bc.TS_FMT),
               "pid": pids[i % len(pids)], "name": "X",
               "meal": bc.MEALS[i % len(bc.MEALS)], "status": "COLLECTED",
               "machine": "BENCH-STATION"}
        t = time.perf_counter_ns()
        store.scan_persist(rec)
        s_persist.append((time.perf_counter_ns() - t) / 1e6)

    # --- unsynced_count(): called on EVERY scan for the status badge ---
    s_cnt = []
    for _ in range(min(ITER, 3000)):
        t = time.perf_counter_ns()
        store.unsynced_count()
        s_cnt.append((time.perf_counter_ns() - t) / 1e6)

    table([
        Stat("resolve() exact id hit", s_exact),
        Stat("resolve() numeric-suffix hit", s_num),
        Stat("resolve() miss (rejected barcode)", s_miss),
        Stat("person_fields()", s_pf),
        Stat("scan_decide() new -> COLLECTED", s_dec_new),
        Stat("scan_decide() repeat -> ALREADY", s_dec_dup),
        Stat("scan_persist() SQLite fsync commit", s_persist),
        Stat("unsynced_count() (per-scan badge)", s_cnt),
    ])

    p99_persist = pct(s_persist, .99)
    say(f"\nSQLite is configured synchronous=FULL, so every scan costs a real "
        f"fsync.\nMeasured p99 = {p99_persist:.2f} ms on THIS disk. This is "
        f"off the perceived\npath (it runs after update_idletasks), so it "
        f"caps sustained throughput,\nnot felt latency: ceiling is "
        f"~{1000.0 / max(p99_persist, 1e-6):.0f} scans/sec per station.")

    if p99_persist > 50:
        verdict("WARN", "durability",
                f"fsync p99 {p99_persist:.1f} ms is slow for this disk — "
                "sustained scan rate is capped near "
                f"{1000 / p99_persist:.0f}/s.")
    else:
        verdict("PASS", "durability",
                f"fsync p99 {p99_persist:.1f} ms — durability cost is fine.")

    if pct(s_dec_new, .99) > 1.0:
        verdict("WARN", "scan-decide",
                "in-memory decide p99 exceeded 1 ms.")
    else:
        verdict("PASS", "scan-decide",
                f"in-memory decide p99 {pct(s_dec_new, .99):.3f} ms.")

    dq._drop_thread_conn()

#  PHASE 2 — local queue scaling as the DB grows over a multi-day event
# --------------------------------------------------------------------------- #
def phase2_queue_scaling() -> None:
    rule("PHASE 2 — SQLite queue cost as the local DB grows")
    say("Does a big scan_queue.db slow anything down mid-event?\n")

    sizes = [1_000, 5_000] if QUICK else [1_000, 10_000, 50_000, 150_000]
    rows_hdr = (f"{'db rows':>10} {'pending(100)':>14} {'pending(ALL)':>14} "
                f"{'pending_count':>14} {'all_rows()':>12} {'file MB':>9}")
    say(rows_hdr)
    say("-" * len(rows_hdr))

    warned = False
    for n in sizes:
        path = os.path.join(SANDBOX, f"scale_{n}.db")
        dq = bc.DurableQueue(path)
        # Bulk-load fast on a private connection (setup, not measurement).
        con = sqlite3.connect(path)
        con.execute("PRAGMA synchronous=OFF")
        con.executemany(
            "INSERT INTO scans(scan_id, ts, pid, name, meal, status, machine,"
            " synced) VALUES (?,?,?,?,?,?,?,?)",
            [(f"S{n}-{i:09d}", datetime(2026, 8, 1).strftime(bc.TS_FMT),
              f"ISMUN-{(i % 1000) + 1:04d}", "Name", bc.MEALS[i % 12],
              "COLLECTED", "BENCH", 1 if i < n - 250 else 0)
             for i in range(n)])
        con.commit()
        con.close()

        def med(fn, reps=7):
            xs = []
            for _ in range(reps):
                t = time.perf_counter_ns()
                fn()
                xs.append((time.perf_counter_ns() - t) / 1e6)
            return statistics.median(xs)

        t_p100 = med(lambda: dq.pending(100))
        t_pall = med(lambda: dq.pending(1_000_000_000))
        t_cnt = med(lambda: dq.pending_count())
        t_all = med(lambda: dq.all_rows(), reps=3)
        mb = os.path.getsize(path) / 1e6

        say(f"{n:>10,} {t_p100:>13.3f} {t_pall:>13.3f} {t_cnt:>13.3f} "
            f"{t_all:>11.3f} {mb:>8.2f}")

        if t_cnt > 5.0 and not warned:
            warned = True
        dq._drop_thread_conn()

    say("\nNotes:")
    say("  pending(100)   — the flusher's batch read. Indexed, stays flat.")
    say("  pending(ALL)   — called by reconcile() every 5 s with a limit of")
    say("                   1,000,000,000. Only UNSYNCED rows match, so it")
    say("                   stays small in practice; it only blows up if the")
    say("                   network is down for a long stretch.")
    say("  pending_count()— runs on EVERY scan for the status badge. It is a")
    say("                   COUNT(*) over the partial index, so it is O(1)-ish")
    say("                   in unsynced rows, not total rows.")
    say("  all_rows()     — startup only (bootstrap_local). Grows linearly;")
    say("                   this is your app-launch cost, not a scan cost.")

    if warned:
        verdict("WARN", "queue",
                "pending_count() exceeded 5 ms at scale — it runs on every "
                "scan.")
    else:
        verdict("PASS", "queue",
                "local queue reads stay flat as the DB grows.")


# --------------------------------------------------------------------------- #
#  PHASE 3 — reconcile() scaling: the sheet-as-truth rebuild
# --------------------------------------------------------------------------- #
def phase3_reconcile() -> None:
    rule("PHASE 3 — reconcile() vs Scan Log size  [the main scaling risk]")
    say("The poller does a FULL sheet read + full state rebuild every 5 s.")
    say("Cost grows linearly with total rows ever logged. Measured below,")
    say("with the lock-hold time isolated (that is what can block a scan).\n")

    people = 400 if QUICK else 1200
    sizes = ([500, 2_000, 5_000] if QUICK
             else [500, 2_000, 5_000, 10_000, 20_000, 40_000])
    reps = 3 if QUICK else 6

    store, dq, _ = fresh_store(people)
    lk: InstrumentedRLock = store.lock

    hdr = (f"{'sheet rows':>11} {'reconcile p50':>14} {'p99':>9} "
           f"{'lock hold p99':>14} {'changed cells':>14} {'payload KB':>11}")
    say(hdr)
    say("-" * len(hdr))

    worst_lock = 0.0
    worst_total = 0.0
    worst_payload = 0.0
    for n in sizes:
        rows = make_sheet_rows(people, n)
        payload_kb = sum(len(c) for r in rows for c in r) / 1024.0
        # Prime once so we measure steady-state (no first-run cold cost).
        store.reconcile(rows)
        lk.reset()
        lk.enabled = True
        durations, changed_n = [], 0
        for _ in range(reps):
            t = time.perf_counter_ns()
            changed = store.reconcile(rows)
            durations.append((time.perf_counter_ns() - t) / 1e6)
            changed_n = len(changed)
        lk.enabled = False
        hold_p99 = pct([h / 1e6 for h in lk.holds_ns], .99)
        worst_lock = max(worst_lock, hold_p99)
        worst_total = max(worst_total, pct(durations, .99))
        worst_payload = max(worst_payload, payload_kb)
        say(f"{n:>11,} {pct(durations, .5):>13.2f} {pct(durations, .99):>8.2f} "
            f"{hold_p99:>13.2f} {changed_n:>14,} {payload_kb:>10.1f}")
        del rows
        gc.collect()

    say("\nInterpretation:")
    say("  * reconcile p99 is the poller thread's work — it does NOT block")
    say("    the UI except for the portion spent holding store.lock.")
    say("  * 'lock hold p99' IS the blocking window. A scan that lands inside")
    say("    it waits that long before scan_decide() can proceed.")
    say("  * 'payload KB' is what read_log_all() pulls from Google every 5 s.")

    if worst_lock > 16.0:
        verdict("WARN", "reconcile",
                f"reconcile holds store.lock up to {worst_lock:.1f} ms at the "
                "largest tested log size — a scan landing in that window "
                "stalls by that much.")
    else:
        verdict("PASS", "reconcile",
                f"reconcile lock-hold p99 stays at {worst_lock:.1f} ms — "
                "below one 60 Hz frame.")

    if worst_total > 1000:
        verdict("WARN", "reconcile",
                f"reconcile total p99 {worst_total:.0f} ms approaches the 5 s "
                "poll interval.")

    say(f"\nGoogle recommends keeping a request payload under 2 MB. The")
    say(f"largest log size tested here ({sizes[-1]:,} rows) pulls "
        f"{worst_payload / 1024:.2f} MB")
    say("EVERY 5 SECONDS, on every station, forever.")
    if worst_payload / 1024 > 2.0:
        verdict("WARN", "payload",
                f"at {sizes[-1]:,} logged rows the 5 s poll pulls "
                f"{worst_payload / 1024:.1f} MB — past Google's 2 MB "
                "guidance. Estimate your real row count before the event.")
    else:
        verdict("PASS", "payload",
                f"largest tested poll payload {worst_payload / 1024:.2f} MB — "
                "inside Google's 2 MB guidance.")

    dq._drop_thread_conn()


# --------------------------------------------------------------------------- #
#  PHASE 4 — concurrency: does a background poll stall a scan?
# --------------------------------------------------------------------------- #
def phase4_contention() -> None:
    rule("PHASE 4 — lock contention: scanning WHILE the poller reconciles")
    say("This is the realistic worst case — a delegate scans at the exact")
    say("moment the 5 s poll rebuilds state from a large sheet.\n")

    people = 400 if QUICK else 1200
    sheet_n = 5_000 if QUICK else 25_000
    scans = 300 if QUICK else 900

    store, dq, _ = fresh_store(people)
    pids = store.roster_ids()
    rows = make_sheet_rows(people, sheet_n)
    store.reconcile(rows)

    stop = threading.Event()
    recon_count = [0]

    def poller_loop():
        while not stop.is_set():
            store.reconcile(rows)
            recon_count[0] += 1
            time.sleep(0.01)

    # Baseline: no background reconcile.
    base = []
    for i in range(scans):

# --------------------------------------------------------------------------- #
#  PHASE 5 — Tk: VirtualBoard rendering vs roster size
# --------------------------------------------------------------------------- #
def _kill_app(app) -> int:
    """Cancel every pending 'after' timer, then destroy the root.

    barcode.py schedules an after(900) flash-decay per scan plus a repeating
    after(100) _drain_ui. Destroying the root without cancelling them leaves
    orphaned Tcl callbacks that fire against the NEXT root and spam
    'invalid command name'. Returns how many timers were outstanding — a
    useful number in its own right (see PHASE 10)."""
    n = 0
    try:
        pending = app.tk.call("after", "info")
        ids = app.tk.splitlist(pending) if isinstance(pending, str) else pending
        n = len(ids)
        for aid in ids:
            try:
                app.after_cancel(aid)
            except Exception:                             # noqa: BLE001
                pass
    except Exception:                                     # noqa: BLE001
        pass
    try:
        _kill_app(app)
    except Exception:                                     # noqa: BLE001
        pass
    return n


def _make_app(n_people: int):
    """Build a real App on a real Tk root, mapped and sized."""
    store, dq, path = fresh_store(n_people)
    app = bc.App(store)
    app.geometry("1500x800")
    app.update_idletasks()
    app.update()
    bv = app.boardview
    # Force slot allocation if <Configure> has not delivered a real height.
    if bv._nslots == 0 or bv.body.winfo_height() <= 1:
        bv.body.configure(height=700)
        app.update_idletasks()
        bv._on_resize()
        app.update()
    return app, store, dq, path

def cleanup_and_verify() -> None:
    rule("CLEANUP & INTEGRITY VERIFICATION")

    created = []
    for root, dirs, files in os.walk(SANDBOX):
        for f in files:
            created.append(os.path.join(root, f))
    total_mb = sum(os.path.getsize(p) for p in created if os.path.exists(p))
    total_mb /= 1e6
    say(f"Sandbox           : {SANDBOX}")
    say(f"Files created     : {len(created)}  ({total_mb:.1f} MB)")

    if ARGS.keep:
        say("KEPT on request (--keep). Delete it yourself when done.")
    else:
        shutil.rmtree(SANDBOX, ignore_errors=True)
        gone = not os.path.exists(SANDBOX)
        say(f"Sandbox removed   : {'YES' if gone else 'NO — remove manually!'}")
        if not gone:
            verdict("FAIL", "cleanup", f"could not remove {SANDBOX}")

    fp_after = fingerprint(PROTECTED)
    changed = [p for p in PROTECTED if FP_BEFORE.get(p) != fp_after.get(p)]
    say("\nProduction files checked for tampering:")
    for p in PROTECTED:
        state = FP_BEFORE.get(p)
        mark = "CHANGED!" if p in changed else (
            "unchanged" if state else "absent (nothing to protect)")
        say(f"  [{mark:^32}] {p}")
    if changed:
        verdict("FAIL", "safety",
                f"{len(changed)} production file(s) were modified: {changed}")
    else:
        verdict("PASS", "safety",
                "no production file was created, modified, or deleted.")

    say(f"\nReal SheetClient constructed: "
        f"{'NO — zero network calls made.' if bc.SheetClient is _ForbiddenSheetClient else 'YES — INVESTIGATE'}")


def summary() -> None:
    rule("SUMMARY")
    order = {"FAIL": 0, "WARN": 1, "PASS": 2}
    for lvl, area, msg in sorted(VERDICTS, key=lambda v: order.get(v[0], 3)):
        say(f"  [{lvl:^4}] {area:<12} {msg}")
    fails = sum(1 for v in VERDICTS if v[0] == "FAIL")
    warns = sum(1 for v in VERDICTS if v[0] == "WARN")
    say("")
    if fails:
        say(f"RESULT: {fails} FAIL, {warns} WARN — investigate above.")
    elif warns:
        say(f"RESULT: {warns} WARN, no failures — usable, see notes.")
    else:
        say("RESULT: all checks passed.")
